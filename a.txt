-- Table: Libraries CREATE TABLE Libraries ( LibraryId INT IDENTITY(1,1) PRIMARY KEY, Name NVARCHAR(150) NOT NULL, Address NVARCHAR(250) NOT NULL, MaximumCapacity INT NOT NULL, CreatedAt DATETIME2 DEFAULT GETDATE(), UpdatedAt DATETIME2, CONSTRAINT CK_Libraries_MaxCap CHECK (MaximumCapacity >= 0) ); 
-- Table: Books CREATE TABLE Books ( BookId INT IDENTITY(1,1) PRIMARY KEY, Title NVARCHAR(200) NOT NULL CHECK (LEN(LTRIM(RTRIM(Title))) > 0), Author NVARCHAR(150) NOT NULL, Price DECIMAL(10,2) NOT NULL CHECK (Price >= 0), Category NVARCHAR(100) NOT NULL, LibraryId INT NOT NULL, CreatedAt DATETIME2 DEFAULT SYSDATETIME(), UpdatedAt DATETIME2, CONSTRAINT FK_Books_Libraries FOREIGN KEY (LibraryId) REFERENCES Libraries(LibraryId) ON DELETE CASCADE );
